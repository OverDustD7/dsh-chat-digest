      //   为什么要带：**压缩策略是建会话时定死的**（`SessionHeader.agentPreset` 持久化 +
      //     `assertPresetUnchanged` 事后不许改），用默认 preset 建的会话永远吃 `retainRatio` 0.16
      //     （THU 200k ⇒ 只逐字保留 32k，压一次就丢细节）。
      //   降级保护：preset 服务不可用/查不到/建会话报错 ⇒ **退回不带 preset 再建一次**。
      //     调优是可选件，采集线绝不能因为它整个挂掉；降级只记步骤、不进 `S.lastError`（免得报假警）。
const AGENT_PRESET = (typeof CF.agentPreset === 'string') ? CF.agentPreset : '';
      let presetOk = null;   // null=还没探过；探到一次就缓存，避免每建一次会话都问一遍
      const presetUsable = async () => {
        if (presetOk !== null) return presetOk;
        try {
          const ap = ctx.get('agentPresets');
          if (!ap || !ap.resolve) { presetOk = false; return presetOk }
          const r = await ap.resolve(AGENT_PRESET);
          presetOk = !!(r && String(r.id) === AGENT_PRESET);
        } catch (e) { presetOk = false }
        return presetOk;
      };
      const createSession = async (sc, base, steps) => {
        const tag = (steps && steps.push) ? (s) => steps.push(s) : () => {};
        if (await presetUsable()) {
          try {
            const made = await sc.create(Object.assign({}, base, { agentPreset: AGENT_PRESET }));
            tag('preset:' + AGENT_PRESET);
            return made;
          } catch (e) {
            presetOk = false;   // 记忆化关掉，后面所有建会话都走降级分支
            tag('preset-fallback:' + String((e && e.message) || e).slice(0, 40));
          }
        } else tag('preset:none');
        return await sc.create(base);
      };
      const checkUsable = async (sc, id, steps, wsIds, arch) => {
        // **首选工作区成员表**（`wc.create().workspace.sessionIds`，不分页、最权威）。
        // 2026-09-17 晚实测出锅：这里原来只看 `sc.list()`，而它是**分页**的 ⇒ 槽会话不在第一页就被判
        //   `stale:list-missing` ⇒ 判死 ⇒ 认领失败 ⇒ **又新建一个主 agent 并灌 WAKE_TEXT**（两个主 agent 同时跑）。
        //   这与 `liveOk` 那次是同一个坑 —— 判活一律以成员表为准，`sc.list()` 只能当兜底。
        try {
          const list = (wsIds || []).map(String);
          const dead = (arch || []).map(String);
          if (list.length && list.indexOf(String(id)) >= 0 && dead.indexOf(String(id)) < 0) {
            steps.push('alive:ws-member'); return { alive: true, needWake: false };
          }
        } catch (e) {}
        const sum = await sessionSummary(sc, id);
        if (sum === undefined) { steps.push('alive:list-unavailable'); return { alive: true, needWake: false } }
        if (sum === null) { steps.push('stale:list-missing'); return { alive: false, needWake: false } }
        if (sum.blank === true) { steps.push('alive:blank-needs-wake'); return { alive: true, needWake: true } }
        steps.push('alive:ok');
        return { alive: true, needWake: false };
      };

      //: DSH 家的位置（会话投影缓存在这里）。默认 $DSH_HOME，没设就 %USERPROFILE%\.dsh —— 不写死用户名。
      const DSH_HOME = (typeof CF.dshHome === 'string' && CF.dshHome) ? CF.dshHome
        : (ENV.DSH_HOME ? ENV.DSH_HOME : ((ENV.USERPROFILE || 'C:') + '\\.dsh'));
      const PROJ_DIR = DSH_HOME + '\\storages\\session_projcache\\sessions\\';
      const sessionMeta = async (fs2, sid) => {
        try {
          const txt = await fs2.readText(await fs2.resolve(PROJ_DIR + sid + '.json'));
          const j = JSON.parse(txt);
          const rec = (j && j.record) || {};
          const ident = rec.identity || {};
          let title = '';
          try { title = String(rec.rows.title.val || '') } catch (e) {}
          let blank = false;
          try { blank = rec.rows.sessionListMetadata.val.blank === true } catch (e) {}
          return { createdAt: Number(ident.createdAt) || 0, title: title, blank: blank };
        } catch (e) { return null }
      };

      // 活成员闸门（2026-09-13 加，用户报"为啥出现两个主 agent"）：
      // 上面 adoptExistingMain 的第一条判据（锚点）和"活的成员表"是两套口径，
      // 而活成员表才是侧栏真正渲染的东西。既然某个活成员已经是同名主 agent，就绝不该再 create。
      // ⚠ 必须同时排除"已归档"的：成员表里**会**保留已归档的会话（实测某个 id 既在 members 又在
      // archivedSessionIds 里）。上一版只查 members，于是把一个已归档的会话又认领回来，
      // 造成"认领 → 下轮判 stale → 再认领"的死循环（/ss 返回的 sessionId 一直是那个已归档的）。
      // 主 agent 的标题现在有**三种形态**（2026-09-17 晚）：通用名 + 两条槽的带后缀名。
      // ⚠ 这里只认通用名会出大事（当晚实测出锅）：槽标题改成带后缀后，`ensureSession` 就**找不到"主 agent"**了
      //   ⇒ 判定"没有活的主 agent" ⇒ **又新建一个并灌 WAKE_TEXT** ⇒ 两个主 agent 同时跑。
      //   所以凡是要判"这条会话是不是主 agent"，一律用这个函数，不要直接和 SESSION_TITLE 比。
      const isMainTitle = (t) => {
        const s = String(t || '');
        const bases = [SESSION_TITLE].concat(typeof LEGACY_TITLES !== 'undefined' ? LEGACY_TITLES : []);
        return bases.some(function (b) {
          return s === b || (typeof ROUTES !== 'undefined' ? ROUTES : []).some(function (r) { return s === b + '（' + r.label + '）' });
        });
      };

      const adoptLiveByTitle = async (sc, wsIds, arch, steps) => {
        const list = (wsIds || []).map(String);
        if (!list.length) return '';
        const dead = (arch || []).map(String);
        const cands = [];
        for (let i = 0; i < list.length && i < 60; i++) {
          const id = list[i];
          if (!id.startsWith('session-')) continue;
          if (dead.indexOf(id) >= 0) continue;              // 已归档：绝不再认领
          const meta = await sessionMeta(ctx.get('fs'), id);
          if (meta && isMainTitle(meta.title)) cands.push({ id: id, createdAt: meta.createdAt || 0 });
        }
        if (!cands.length) return '';
        cands.sort((a, b) => a.createdAt - b.createdAt);
        steps.push('adopt:live-title:' + cands[0].id.slice(0, 12));
        return cands[0].id;
      };

      // A23（2026-09-25 用户问"为什么多了一堆主agent"）：**把"多出来的同标题主 agent"扫出来给面板看**。
      //   实测：用户问这话时插件**自己也不知道**有这事 —— 它只认 `S.sessionId` 那一条，其余同标题的
      //   活会话（当时 3 条）在插件眼里不存在，只能靠我翻 `~/.dsh/storages/workspace.json` 反推。
      //   现在每 5 分钟扫一次工作区成员，除当前会话与两个槽之外的、标题是主 agent 的，列进
      //   `/state.extraMains`；面板显示并可一键归档（`POST /chat-feed/api/archive`）。
      const scanExtraMains = async () => {
        try {
          const wc = ctx.get('workspaceController');
          if (!wc || !wc.create) return;
          const made = await race(wc.create({ path: AGENT_CWD }), 4000);
          const ids = ((made && made.workspace && made.workspace.sessionIds) || []).map(String);
          const arch = await readArchived(wc, []);
          const dead = (arch || []).map(String);
          const cur = String(S.sessionId || ''), thu = String(S.sessThu || ''), par = String(S.sessParatera || '');
          const extra = [];
          for (let i = 0; i < ids.length && i < 60; i++) {
            const id = ids[i];
            if (!id.startsWith('session-')) continue;
            if (dead.indexOf(id) >= 0) continue;
            if (id === cur || id === thu || id === par) continue;
            const meta = await sessionMeta(ctx.get('fs'), id);
            if (meta && isMainTitle(meta.title)) {
              extra.push({ id: id, title: String(meta.title || ''), createdAt: Number(meta.createdAt) || 0 });
            }
          }
          extra.sort((a, b) => a.createdAt - b.createdAt);
          const before = JSON.stringify(S.extraMains || []);
          S.extraMains = extra; S.extraMainsAt = Date.now();
          if (JSON.stringify(extra) !== before) {
            // **只在 ≥2 条时才写 note / 提示**（用户 2026-09-25 定）：留 1 条当备份是正常选择，
            //   不该让面板一直挂着一张"有 1 条多余的"卡片（面板那边用同一门槛）。
            if (extra.length >= 2) S.note = '工作区里有 ' + extra.length + ' 条多余的同标题主 agent（面板可一键归档）';
            await saveState();
          }
        } catch (e) { /* 扫描失败不致命：它只是"看得见"的问题 */ }
      };

      const adoptExistingMain = async (sc, wc, wsIds, arch, steps) => {        const alive = [];
        const consider = async (tag, id) => {
          if (!id || !String(id).startsWith('session-')) return;
          if (arch && arch.indexOf(String(id)) >= 0) { steps.push('adopt:' + tag + ':archived'); return }
          const sum = await sessionSummary(sc, id);
          if (!sum || !sum.sessionId) { steps.push('adopt:' + tag + ':missing'); return }
          const meta = await sessionMeta(ctx.get('fs'), String(id));
          alive.push({ tag: tag, id: String(id), meta: meta, createdAt: (meta && meta.createdAt) || 0, title: (meta && meta.title) || '' });
        };
        if (S.mainSessionId) await consider('anchor', String(S.mainSessionId));
        if (S.sessionId && String(S.sessionId) !== String(S.mainSessionId)) await consider('from-state', String(S.sessionId));
        if (!alive.length && wsIds && wsIds.length) {
          for (let i = 0; i < wsIds.length && i < 40; i++) await consider('ws', String(wsIds[i]));
        }
        const titled = alive.filter((x) => isMainTitle(x.title)).sort((a, b) => a.createdAt - b.createdAt);
        if (titled.length) { steps.push('adopt:' + titled[0].tag + ':' + titled[0].id.slice(0, 12)); return titled[0].id }
        const any = alive.slice().sort((a, b) => a.createdAt - b.createdAt);
        if (any.length) { steps.push('adopt:earliest:' + any[0].id.slice(0, 12)); return any[0].id }
        return '';
      };

      // ⚠ 并发保护（2026-09-13 加，用户问"都归档了你不会自己新建一个吗"）：
      // 认领/新建这套逻辑在"工作区里一个活会话都没有"时**必须新建**（这次就是全部归档 → 它自己新建了
      // session-28a5）。但 /ss 会被好几个入口同时打到（UI 的 openPanel + boot、/chat、/tell），
      // 而"没有活会话"这个状态在两次 create 之间是**不变的** → 两个并发 /ss 会各建一个主 agent，
      // 就是历史那个"反复新建"的根因。这里用一个进程内的 mutex：同一时刻只跑一次 ensureSession，
      // 重复调用直接复用同一个 promise。
      let ensureInFlight = null;
      const ensureSession = async (wantSeed) => {
        if (ensureInFlight) return ensureInFlight;
        const run = (async () => {
          const sc = ctx.get('sessionController'), wc = ctx.get('workspaceController');
          if (!sc || !sc.create) return { ok: false, error: 'sessionController 不可用' };
          const steps = ['load:' + (await loadState())];
        const loadedId = String(S.sessionId || '');   // 用于最后判断"会话 id 是否变了"，变了要落盘
        let wsId = '', wsIds = null, wsTitle = '';
        if (wc && wc.create) {
          try {
            const made = await race(wc.create({ path: AGENT_CWD }), 4000);
            const w = made && made.workspace;
            if (w && w.workspaceId) {
              wsId = String(w.workspaceId); wsIds = (w.sessionIds || []).map(String); wsTitle = String(w.title || '');
              steps.push(made.created ? 'ws:created' : 'ws:found');
              try {
                if (wsTitle === WORKSPACE_TITLE) steps.push('ws:title-ok');
                // 这个工作区是**按 AGENT_CWD 的路径**查出来的 ⇒ 它就是我们自己的，标题也归我们管；
                // 别人的工作区根本走不到这里。2026-09-25 改名后老工作区还叫旧名，只改「新建」的
                // 会让侧栏永远挂着一个旧名 —— 所以标题不一致就改回来。
                else if (wc.rename) { await wc.rename({ workspaceId: w.workspaceId, title: WORKSPACE_TITLE }); steps.push('ws:renamed:' + String(wsTitle).slice(0, 12) + '->' + WORKSPACE_TITLE) }
                else steps.push('ws:title-kept:' + String(wsTitle).slice(0, 16));
              } catch (e) { steps.push('ws:rename-err') }
            }
          } catch (e) { steps.push('ws:err') }
        }
        if (S.sessionId) {
          const id = String(S.sessionId);
          const arch = await readArchived(wc, steps);
          if (arch && arch.indexOf(id) >= 0) { steps.push('stale:archived'); S.sessionId = '' }
          else if (wsIds && wsIds.indexOf(id) < 0) {
            // 成员表可能滞后（刚新建/刚切工作区）→ 先用"活的会话列表"复核一次再决定是否清空。
            const sum2 = await sessionSummary(sc, id);
            if (sum2 && sum2.sessionId && sum2.blank !== true) steps.push('wsids:lag-but-alive');
            else { steps.push('stale:not-in-workspace'); S.sessionId = '' }
          }
          if (!S.sessionId) S.seeded = false;
        }
        let created = false, needWake = false;
        if (!S.sessionId) {
          const arch1 = await readArchived(wc, [])
          const adopted = await adoptExistingMain(sc, wc, wsIds, arch1, steps)
          if (adopted) { S.sessionId = adopted; if (!S.mainSessionId) S.mainSessionId = adopted }
          // 活成员闸门（见 adoptLiveByTitle 的注释）：archivedSessionIds 可能过时，
          // 但"工作区里的活成员 + 标题"不会。它命中就认领，绝不 create。
          if (!S.sessionId) {
            const live = await adoptLiveByTitle(sc, wsIds, arch1, steps)
            if (live) { S.sessionId = live; if (!S.mainSessionId) S.mainSessionId = live }
          }
        }
        if (!S.sessionId) {
          try {
            const made = await createSession(sc, wsId ? { workspaceId: wsId } : { cwd: AGENT_CWD }, steps);
            const id = made && made.sessionId ? String(made.sessionId) : '';
            if (!id) return { ok: false, error: 'create 未返回 id', steps };
            S.sessionId = id; created = true; needWake = true; retargetQuote([loadedId], id, steps);
            const archA = await readArchived(wc, []);
            // 锚点为空或指着已归档会话时跟着新会话走（否则 mainSessionId 会停在死会话上）
            if (!S.mainSessionId || (archA && archA.indexOf(String(S.mainSessionId)) >= 0)) { S.mainSessionId = id; steps.push('anchor-set:' + id.slice(0, 12)) }
            steps.push('created:' + id.slice(0, 12)); steps.push('persist:' + (await saveState()));
          } catch (e) { return { ok: false, error: 'create 失败：' + String((e && e.message) || e), steps } }
        } else {
          // 归档表要在体检**之前**取（checkUsable 要用它排除"成员表里含已归档会话"这个已知情形）
          const arch2 = await readArchived(wc, [])
          const chk = await checkUsable(sc, S.sessionId, steps, wsIds, arch2);
          if (!chk.alive) { steps.push('reused:dead'); S.sessionId = ''; }
          else {
            steps.push('reused:' + S.sessionId.slice(0, 12)); needWake = !!chk.needWake;
            if (!S.mainSessionId) { S.mainSessionId = String(S.sessionId); await saveState() }
            // 锚点指向一个已归档的会话时，把锚点换成当前这个活着的（2026-09-13 加）。
            // 不换的话每次 /ss 都要先跑一遍"锚点已归档"的判定；而且锚点是"永不失效"语义，
            // 指着一个死会话本身就是坏数据。
            if (S.mainSessionId && String(S.mainSessionId) !== String(S.sessionId) && arch2.indexOf(String(S.mainSessionId)) >= 0) {
              steps.push('anchor-rebase:' + String(S.mainSessionId).slice(0, 12) + '->' + String(S.sessionId).slice(0, 12));
              S.mainSessionId = String(S.sessionId); await saveState();
            }
          }
          if (!S.sessionId) {
            // **先认领、再新建**（与上面那条分支同样的判据）。
            // 这条分支以前直接 create —— 于是"当前会话被判失效"（归档/移出工作区/体检判死）
            // 就会多造一个主 agent，是"反复新建主 agent"的根因。
            const adopted2 = await adoptExistingMain(sc, wc, wsIds, arch2, steps)
            if (adopted2) { S.sessionId = adopted2; if (!S.mainSessionId) S.mainSessionId = adopted2 }
            // 同上：活成员闸门（必须带上 arch2 —— 成员表里也含已归档的会话）
            if (!S.sessionId) {
              const live2 = await adoptLiveByTitle(sc, wsIds, arch2, steps)
              if (live2) { S.sessionId = live2; if (!S.mainSessionId) S.mainSessionId = live2 }
            }
          }
          if (!S.sessionId) {
            try {
              const made = await createSession(sc, wsId ? { workspaceId: wsId } : { cwd: AGENT_CWD }, steps);
              const id = made && made.sessionId ? String(made.sessionId) : '';
              if (!id) return { ok: false, error: 'create 未返回 id', steps };
              S.sessionId = id; created = true; needWake = true; retargetQuote([loadedId], id, steps);
              const archA = await readArchived(wc, []);
              // 锚点为空或指着已归档会话时跟着新会话走（否则 mainSessionId 会停在死会话上）
              if (!S.mainSessionId || (archA && archA.indexOf(String(S.mainSessionId)) >= 0)) { S.mainSessionId = id; steps.push('anchor-set:' + id.slice(0, 12)) }
              steps.push('created:' + id.slice(0, 12)); steps.push('persist:' + (await saveState()));
            } catch (e) { return { ok: false, error: 'create 失败：' + String((e && e.message) || e), steps } }
          }
        }
        try { if (sc.resolveAgent) await sc.resolveAgent(S.sessionId); steps.push('resume:ok') } catch (e) { steps.push('resume:err') }
        if (sc.rename) {
          // 2026-09-17 下午：**别把"槽"的标题抹平**。`adoptRouteByTitle` 靠标题后缀（＝那条线配置的 `label`）认回会话，
          //   而这里原来**无条件**把当前会话改名成通用标题 ⇒ 槽的标记被抹掉
          //   （实测：THU 槽 `session-<示例>` 的标题就变成了无后缀的 `聊天摘要 · 主 Agent`，
          //    侧栏里一行行同名、分不清谁是槽；一旦槽丢了还会因为认不回标题而**再造一个会话**）。
          //   现在：当前会话属于某个槽 ⇒ 用那条线路的标题；否则才用通用标题。
          const slotRoute = (String(S.sessionId) === String(S.sessThu || '')) ? R0
                          : (String(S.sessionId) === String(S.sessParatera || '')) ? R1 : '';
          const wantTitle = slotRoute
            ? routeTitle(slotRoute)
            : SESSION_TITLE;
          if (S.renamedAs === wantTitle && !created) { steps.push('rename:skipped') }
          else {
            try { await sc.rename({ sessionId: S.sessionId, title: wantTitle }); S.renamedAs = wantTitle; steps.push('rename:ok:' + (slotRoute || 'main')) }
            catch (e) { steps.push('rename:err') }
          }
        }
        // 唤醒只对"这个具体会话"发一次（2026-09-13 用户报"每点一次就多发一次提示词"）。
        // S.seeded 是全局布尔，会话一旦被判失效就被置回 false → 下一次 /ss 会给"新认领的旧会话"
        // 再灌一遍 wakeText。改成按会话 id 记账：S.seededFor === 当前会话 id 就绝不重发。
        if ((wantSeed || needWake) && !S.seeded && S.seededFor !== String(S.sessionId)) {
          try {
            await sc.prompt({ requestId: 'seed-' + Date.now(), sessionId: S.sessionId, mode: 'queue', content: [{ type: 'text', text: WAKE_TEXT }] }, FAKE_SIGNAL);
            S.seeded = true; S.seededFor = String(S.sessionId); steps.push(needWake && !wantSeed ? 'wake:ok' : 'seed:ok'); await saveState();
          } catch (e) { steps.push('seed:err'); noteError('seed', e) }
        } else steps.push(S.seededFor === String(S.sessionId) && !S.seeded ? 'seed:alreadysent' : 'seed:skipped');
        // 会话 id 变了就落盘。以前"被归档 → 认领锚点 → reused"这条路径从不 saveState，
        // 于是内存里已是锚点、state.json 还留着那个已归档的 id（下次 loadState 会读回它，白绕一圈）。
        if (String(S.sessionId || '') !== loadedId) { try { steps.push('persist:' + (await saveState())) } catch (e) {} }
          return { ok: true, sessionId: S.sessionId, sessionTitle: SESSION_TITLE, steps, needWake, lastError: S.lastError };
        })();
        // 无论成败都要解开锁；失败结果也让并发的那些拿到（它们本来就是同一次请求的副本）
        ensureInFlight = run.catch((e) => ({ ok: false, error: String((e && e.message) || e), steps: ['ensure:throw'] }))
          .then((r) => { ensureInFlight = null; return r });
        return ensureInFlight;
      };

      // ── 引用以「组装提示词时注入上下文」的形式送达主 agent（用户 2026-09-15 要求）──

            if (n < 10) { const t = setTimeout(() => { try { ensureBothSlots() } catch (e) {} }, 3000); if (t && t.unref) t.unref() }

        S.slotsInitBusy = true;

          await ensureRouteSession(R1, steps);      // 槽空/失效才新建；已有活着的就复用

          const pk = await pickRoute(S.routePick, 'slots-init');

        } catch (e) { noteError('slots-init', e) } finally { S.slotsInitBusy = false }
      };

      // A10（2026-09-20 审计 H07）：**进入触发入口就占互斥锁**。
      //   旧版只查 `S.busy`，而 `S.busy` 要到 `fireRound` 末尾才置起 —— 中间隔着探测网络、
      //   选会话、读事件、入队（探针实测并发两次 triggerRound ⇒ dispatch 2 次）。
      //   锁放在最外层（手动 / 自动 / 菜单三条路都经过它），异常一律在 finally 释放；
      //   真正的逻辑挪进 `triggerRoundInner`，好让 try/finally 只包一层。
      const triggerRound = async (reason, opts) => {
        const sc0 = ctx.get('sessionController');
        if (!sc0) return { ok: false, error: '没有 sessionController' };
        if (S.busy) return { ok: false, error: '上一轮还在跑', busy: true, note: S.note };
        if (S.triggering) return { ok: false, error: '另一轮触发正在进行中（入口互斥锁已占用）', busy: true, note: S.note };
        S.triggering = { at: Date.now(), reason: String(reason || '') };
        try {
          return await triggerRoundInner(reason, opts);
        } finally {
          S.triggering = null;
        }
      };
      const triggerRoundInner = async (reason, opts) => {
        const sc = ctx.get('sessionController');
        if (!sc) return { ok: false, error: '没有 sessionController' };
        const o = opts || {};
        // 【2026-09-17 用户要求：侧栏「获取」下拉里点了哪条线路，就现场探哪条】
        //   通了才触发；**不通就直接回失败**（前端把按钮闪成红底白字「获取失败」），
        //   **绝不写 `lastCollectAt`/`collectDay`** ⇒ 上次采集时间保持原样（这正是用户要的语义）。
        //   注意：这条路**不看 gwPolicy**（用户已经明确选了线路，不再替他决定花不花钱）。
        if (o.probeRoute) {
          const only = (String(o.probeRoute) === String(R1)) ? R1 : R0;
          const pr0 = await pluginProbe(only);
          if (!pr0.ok) {
            S.gwDownAt = Date.now();
            S.gwDownReason = String(pr0.err || ('HTTP ' + pr0.status) || 'probe-failed');
            S.note = '探测 ' + only + ' 不通 ⇒ 这一轮没跑（' + cstHM() + '，采集时间未改）';
            await saveState();
            return { ok: false, routeDown: true, route: only, probe: pr0, note: S.note,
                     error: '线路 ' + String(only || target) + ' 探测不通'
                            + (pr0.err ? '（' + pr0.err + '）' : pr0.status ? '（HTTP ' + pr0.status + '）' : '')
                            + ' ⇒ 没跑这一轮，上次采集时间不变' };

          if (!pickM || !pickM.ok) return { ok: false, error: (pickM && pickM.error) || '选会话失败', probe: pr0 };
          return await fireRound(sc, reason, pickM, String(o.forceMode || 'full'), pr0);
        }
        // **双会话：唤醒前先探 THU，再选中该用的那个会话**（2026-09-17 用户定案；不再用会话内 selectModel 换供应商）
        const probeRes = o.forceRoute ? { at: Date.now(), ok: (String(o.forceRoute) === String(R0)), status: 0, ms: 0, err: 'manual-force', route: o.forceRoute } : await pluginProbe();
        let mode = String(o.forceMode || 'full');         // 'full' | 'pipeline'
        if (!probeRes.ok) {
          // **网关不通**：按面板策略处理（用户 2026-09-17 定：默认「不跑」）。
          //   注意**不再弹 question 卡片** —— 那类卡片只在"提问那条会话"的输入框里，后台会话你看不到，
          //   而且"问"本身就得先唤醒付费会话；所以**自动**触发那一轮按 `gwPolicy` 处理（默认不跑）。
          const pol = String(S.gwPolicy || 'skip');
          S.gwDownAt = Date.now();
          S.gwDownReason = String(probeRes.err || ('HTTP ' + probeRes.status) || 'probe-failed');
          S.routeNeedsConfirm = false;                    // 保留字段但不再用于"问他"
          if (pol === 'skip' && !o.forceRoute) {
            S.note = '网关不通 ⇒ 按策略未跑（' + cstDay() + '）';
            await saveState();
            // 带上人话的 error：面板（ui.js）把 ok:false 当失败、会读 r.error 显示出来
            return { ok: false, skipped: true, reason: 'gateway-down', policy: pol, probe: probeRes, note: S.note,
                     error: '线路不通 ⇒ 按策略「不通则不跑」没有跑这一轮（可用侧栏「获取」下拉换一条线路手动跑）' };
          }
          mode = (pol === 'pipeline' && !o.forceMode) ? 'pipeline' : mode;
          const pick2 = await pickRoute(R1, 'gw-down:' + pol + (o.forceRoute ? ':manual' : ''));
          if (!pick2 || !pick2.ok) return { ok: false, error: (pick2 && pick2.error) || '切付费会话失败', probe: probeRes };
          return await fireRound(sc, reason, pick2, mode, probeRes);
        }
        // A27（2026-09-25）：这句 why 原先**写死"THU"** —— 探的却是 `probeRes.route`。
        //   于是探 paratera 成功时会报 `routeWhy = "probe:THU 可达"`，而 `routePick = paratera`
        //   ⇒ 自证字段当场说谎（实测 2026-09-25 /state：routePick=paratera、routeWhy="probe:THU 可达"）。
        //   现按**实际探的那条线**报，并配 H23 探针。
        const pick = await pickRoute(probeRes.route, 'probe:' + probeRes.route + (probeRes.ok ? ' 可达' : ' 不通(' + (probeRes.err || probeRes.status) + ')'));
        if (!pick || !pick.ok) return { ok: false, error: (pick && pick.error) || '选会话失败', steps: pick && pick.steps };
        return await fireRound(sc, reason, pick, mode, probeRes);
      };
      // 真正把提示词发出去（两条路径共用）
      const fireRound = async (sc, reason, pick, mode, probeRes) => {
        const date = cstDay();
        const base = await turnEnds(sc);                 // 触发前的 turn/end 计数 = 基线
        try {
          await sc.prompt({ requestId: 'collect-' + Date.now(), sessionId: S.sessionId, mode: 'queue',
                            content: [{ type: 'text', text: roundPrompt(date, reason, S.lastCollectAt || 0, mode) }] }, FAKE_SIGNAL);
        } catch (e) { noteError('collect-prompt', e); return { ok: false, error: String((e && e.message) || e) } }
        S.busy = true; S.busySince = Date.now(); S.busyBase = base;
        // **记下这一轮到底发进了哪个会话**（2026-09-17 晚加）：这一轮跑起来之后 `S.sessionId` 有可能被
        //   `ensureSession` 换掉（那次"两个主 agent"事故就是这么发生的），而 checkBusy 原来只看 S.sessionId
        //   ⇒ 换了会话就永远收不了尾（busy 卡到 3 小时超时，面板一直"采集中…"、获取被"上一轮还在跑"挡住）。
        S.busyFor = String(S.sessionId || ''); S.busySawOpen = false;
        S.retryCount = 0; S.lastRetryAt = 0;                // 新一轮 ⇒ 续跑计数归零
        // A04（2026-09-20 审计 H06）：**这里绝不推进采集游标**。
        //   旧版在这里写 `S.lastCollectAt = Date.now(); S.collectDay = date;` —— 而此刻只是
        //   "提示词入队成功"，取数还没跑、面板还没被回写。中断/取数失败/没交付都会留下
        //   一个**再也不会被覆盖的空洞**（下一轮从"入队时刻"开始取数）。
        //   现在只记"已请求、未交付"；游标由 `commitCursor()` 在**面板真被回写**时提交。
        S.pendingCollect = { day: date, at: Date.now(), sessionId: String(S.sessionId || ''),
                             reason: String(reason || ''), mode: String(mode || '') };
        // A22：新一轮开始 ⇒ 上一轮那个没人回答的提问作废（它属于上一轮，不该继续在面板上挂着）
        S.pendingAsk = null;
        S.gwRunAt = Date.now(); S.gwRunMode = mode;
        S.note = '已触发 ' + date + ' 一轮（' + (reason === 'auto' ? '自动' : '手动') + '，' + S.routePick + '，mode=' + mode
               + '）；**采集游标等交付成功才前移**';
        await saveState();
        return { ok: true, date, reason, route: S.routePick, mode: mode, sessionId: S.sessionId,
                 steps: (pick && pick.steps) || [], probe: probeRes, note: S.note };
      };

      // ---- 上下文用量探测（2026-09-15 晚新增；判据全部来自会话事件，不猜）----
      //   · 真实窗口：`request/context` 事件带 `contextWindow`（DSH SessionEventMap 里的类型）
      //   · 当前用量：最新一条 `assistant/message` 的 `usage`（inputTokens = 这次请求带进去的上下文）
      //   · 有没有 turn 正在跑：最后一个 turn 只有 start 没有 end ⇒ 正在干活
      const ctxProbe = async (sc) => {
        try {
          const ins = await sc.inspect(S.sessionId, FAKE_SIGNAL);
          const evs = (ins && ins.events) || [];
          let win = 0, tokens = 0, turn = -1, lastTurn = -1, usageSeen = 0;
          const ended = {};
          for (let i = 0; i < evs.length; i++) {
            const e = evs[i]; if (!e || !e.data) continue;
            if (e.type === 'turn/start') lastTurn = Math.max(lastTurn, Number(e.data.turn) || 0);
            if (e.type === 'turn/end') { const t = Number(e.data.turn) || 0; ended[t] = true; lastTurn = Math.max(lastTurn, t) }
            if (e.type === 'request/context' && Number(e.data.contextWindow) > 0) win = Number(e.data.contextWindow);
            if (e.type === 'assistant/message' && e.data.usage && typeof e.data.usage === 'object') usageSeen++;
          }
          for (let i = evs.length - 1; i >= 0; i--) {
            const e = evs[i]; if (!e || e.type !== 'assistant/message' || !e.data) continue;
            const u = e.data.usage; if (!u || typeof u !== 'object') continue;
            // ⚠ 单位必须取 **totalTokens**，不能取 inputTokens+outputTokens（2026-09-15 实测踩到）：
            //   缓存命中的上下文**不算进 inputTokens** —— 实测爆掉那次会话最后一条是
            //   `{inputTokens:258, outputTokens:682, totalTokens:797100, cacheReadTokens:796160}`，
            //   按 input+output 会算出 940（0.09% 占用），而真实占用是 **79.7%**。
            //   所以：优先 totalTokens；拿不到时按 input+cacheRead+output 补算。
            const inp = Number(u.inputTokens), out = Number(u.outputTokens);
            const tot = Number(u.totalTokens), cac = Number(u.cacheReadTokens);
            const sum = (Number.isFinite(inp) ? inp : 0) + (Number.isFinite(cac) ? cac : 0)
                      + (Number.isFinite(out) ? out : 0);
            const t = (Number.isFinite(tot) && tot > 0) ? tot : (sum > 0 ? sum : 0);
            if (t > 0) { tokens = t; turn = Number(e.data.turn) || 0; break }
          }
          return { window: win, tokens, turn, lastTurn, open: lastTurn >= 0 && !ended[lastTurn],
                   events: evs.length, usageSeen };
        } catch (e) { return { err: String((e && e.message) || e), events: 0 } }
      };

      // 回合结束后（或空闲时）检查一次：超阈值就自动刷新会话。
      //   **绝不打断在跑的活**：busy / open turn / 同一轮已刷过 / 冷却期内 → 只记数不动作。
      const checkCtx = async (when) => {
        if (!S.ctxAuto || S.ctxChecking) return null;
        if (S.busy) { S.ctxSkips = (S.ctxSkips || 0) + 1; return null }
        const sc = ctx.get('sessionController');
        if (!sc || !sc.inspect || !S.sessionId) return null;
        // 空闲路径节流（回合结束那次不节流）：inspect 是整会话读回，别每 30 秒来一次
        if (when === 'idle-tick' && S.ctxLastProbeAt && Date.now() - S.ctxLastProbeAt < S.ctxIdleIntervalMs) return null;
        S.ctxChecking = true;
        try {
          S.ctxLastProbeAt = Date.now();
          const p = await ctxProbe(sc);
          if (p.err) { noteError('ctx-probe', p.err); return null }
          if (!p.tokens) return null;
          S.ctxTokens = p.tokens; S.ctxWindow = p.window || 0; S.ctxChecks = (S.ctxChecks || 0) + 1;
          // 阈值按**当前会话实际所在的线路**取（不是按 routePick）：THU 用 ctxRatioThu（**0.7**，见上面的注释：
          //   0.9 会越过 DSH 的压缩线 0.8 ⇒ 死参数），paratera 用 ctxRatio
          const _ratio = ctxRatioOf(routeOfActive());
          S.ctxLimit = p.window > 0 ? Math.floor(p.window * _ratio) : S.ctxFallback;
          S.ctxPct = S.ctxLimit > 0 ? Math.round(p.tokens * 1000 / S.ctxLimit) / 10 : 0;
          if (p.open) { S.ctxSkips = (S.ctxSkips || 0) + 1; await saveState(); return null }
          if (p.tokens < S.ctxLimit) { await saveState(); return null }
          if (p.turn === S.ctxLastRotateTurn) { S.ctxSkips = (S.ctxSkips || 0) + 1; await saveState(); return null }
          if (S.ctxLastRotateAt && Date.now() - S.ctxLastRotateAt < S.ctxCooldownMs) {
            S.ctxSkips = (S.ctxSkips || 0) + 1; await saveState(); return null;
          }
          S.ctxLastRotateTurn = p.turn; S.ctxLastRotateAt = Date.now();
          const why = 'ctx ' + S.ctxTokens + '/' + S.ctxLimit + ' (' + S.ctxPct + '%, ' + when + ')';
          await saveState();
          const r = await rotateSession(why);
          if (r && r.ok) S.ctxRotations = (S.ctxRotations || 0) + 1;
          // A23（2026-09-25）：**把轮换的步骤落盘**。旧版只把 `steps` 当返回值，而定时轮换是后台路径
          //   （`checkCtx` 的返回值被丢掉）⇒ 事后**无法判断**它到底归档了谁、`archiveSession` 有没有报错。
          //   实测代价：用户问"为什么多了一堆主 agent"，我只能推断到"轮换造的"，查不出那两次为什么没归档。
          S.ctxLastRotateWhy = String(why).slice(0, 160);
          S.ctxLastRotateOk = !!(r && r.ok);
          S.ctxLastRotateArchived = String((r && r.archived) || '');
          S.ctxLastRotateErr = String((r && r.error) || '');
          S.ctxLastRotateSteps = (r && Array.isArray(r.steps))
            ? r.steps.slice(0, 24).map((x) => String(x).slice(0, 90))
            : ['err:' + String((r && r.error) || 'no-steps')];
          await saveState();
          return r;
        } finally { S.ctxChecking = false }
      };

      // busy 收尾：轮询主 agent 会话的 turn/end 计数，多了一次就说明这一轮结束了。
      // （不用 ctx.on('api-session/status')：动态插件的 ctx.on 订阅不到宿主事件——历史实测，见 DEV_NOTES）
      const checkBusy = async () => {
        if (!S.busy) return;
        const sc = ctx.get('sessionController'); if (!sc) return;
        // 认"这一轮真正发进去的那个会话"（`busyFor`），没有才退回 `S.sessionId`（老状态）
        const sid = String(S.busyFor || S.sessionId || '');
        if (!sid) return;
        const st = await turnState(sc, sid);
        if (st === null) return;
        // A22：**主 agent 在等你回答**（有未回答的 `ask_user_question`）⇒ 记进 `S.pendingAsk`，
        //   由 `/state` 与面板显示。没有这一条，用户在自动轮里**完全不知道它在等**（2026-09-21 的实况）。
        //   解析器就地写（探针只 exec 本函数的切片，外面的名字在它作用域里不存在）。
        if (st.ask) {
          if (!S.pendingAsk || S.pendingAsk.callId !== st.ask.callId) {
            S.pendingAsk = { callId: st.ask.callId, at: Number(st.ask.at) || Date.now(),
                             questions: Array.isArray(st.ask.questions) ? st.ask.questions : [], stale: false };
            S.note = '主 agent 在等你回答一个问题（' + cstHM() + '）—— 去那条会话回答它，这一轮才会继续';
            await saveState();
          }
        } else if (S.pendingAsk && !S.pendingAsk.stale) {
          S.pendingAsk = null;                            // 已回答（或新一轮开始了）
        }
        if (st.open) S.busySawOpen = true;                  // 看到过"正在干活"
        const done = st.open ? false
          : (S.busySawOpen ? true                            // 看到过 open、现在不 open ⇒ 收尾（重启后也能收）
            : (S.busyFor ? st.ends > (Number(S.busyBase) || 0) // 老判据：turn/end 计数变大了
              : true));                                      // 老状态里没有 busyFor 且没在跑 ⇒ 直接收尾，别卡满 3 小时
        if (done) {
          // A05（2026-09-20 审计 H01）：**先把"这一轮的身份"取出来，再清 busyFor**。
          //   旧版顺序是先 `S.busyFor = ''`、再 `if (abn && ... && S.busyFor)` —— 条件永远为假，
          //   续跑分支**根本进不去**（探针实测注入次数 0，期望 1）。这个 bug 从加续跑那天起就在。
          const roundSid = String(S.busyFor || '');
          S.busy = false; S.busyBase = null; S.busyFor = ''; S.busySawOpen = false; S.lastDoneAt = Date.now();
          // **异常退出 ⇒ 自动续跑**（2026-09-17 用户要求："遇到这种情况你要自动让他继续，检测到他异常退出就说'继续'"）：
          //   判据＝这一轮所在会话**最后一个 turn 的结束原因不是 `completed`**（实测网络断流是
          //   `{kind:'error',error:{code:'TRANSPORT',message:'Stream ended without finish_reason'}}`）。
          //   做法＝往**同一个会话**（`sid`，不一定是 S.sessionId）注入一句「继续」，并把 busy 重新置起，
          //   于是它接着跑、也仍然会被这里收尾 ⇒ 连续失败会被重试上限挡住（见 autoRetry）。
          //   **只对 `error` 续跑，绝不对 `aborted` 续跑** —— 用户手动点「停止生成」的语义就是"别跑了"，
          //   自动说「继续」等于跟他对着干。
          //   **只对"我们自己触发的那一轮"续跑**（`S.busyFor` 非空）：否则会把一条被停掉的旁支会话又"续"活。
          const abn = (st.lastReason === 'error') ? st.lastReason : '';
          if (abn && S.autoRetry !== false && roundSid) {
            const tries = Number(S.retryCount || 0) + 1;
            if (tries <= 3) {
              S.retryCount = tries; S.lastRetryAt = Date.now();
              try {
                const base2 = await turnEnds(sc, sid);
                await sc.prompt({ requestId: 'retry-' + Date.now(), sessionId: sid, mode: 'queue',
                                  content: [{ type: 'text', text: '继续' }] }, FAKE_SIGNAL);
                S.busy = true; S.busySince = Date.now(); S.busyBase = base2;
                S.busyFor = sid; S.busySawOpen = false;
                S.note = '上一轮异常结束（' + abn + '）⇒ 已自动续跑（第 ' + tries + ' 次，' + cstHM() + '）';
                await saveState();
                return;
              } catch (e) { noteError('auto-retry', e) }
            } else {
              S.note = '上一轮异常结束（' + abn + '）且已自动续跑 ' + S.retryCount + ' 次，先停下（' + cstHM() + '）';
              await saveState();
              return;
            }
          }
          S.note = '上一轮已结束（' + cstHM() + '）' + (st.lastReason ? '［' + st.lastReason + '］' : '')
                 + (S.pendingAsk ? '，且**有一个提问没被回答**' : '');

      // 「刷新会话」的**唯一实现**（2026-09-15 晚从 /rotate 路由里抽出来，好让"上下文超阈值自动刷新"复用同一段）：
      //   归档当前主 agent 会话 → 在"聊天摘要"工作区新建 → rebase 锚点 → 改名 → 灌唤醒文案 → persist

      const rotateSession = async (reason, explicitOldId) => {

        if (oldId && String(S.sessionId) === oldId) S.sessionId = '';   // 别再把刚归档的认领回来

          createdId = made && made.sessionId ? String(made.sessionId) : '';

          }
        });
        retargetQuote([oldId], createdId, steps);                       // 待发引用跟着新会话走（原目标已归档）
        try { if (sc.resolveAgent) await sc.resolveAgent(createdId) } catch (e) {}
        try { await sc.rename({ sessionId: createdId, title: SESSION_TITLE }); S.renamedAs = SESSION_TITLE; steps.push('rename:ok') } catch (e) { steps.push('rename:err') }
        try {
          await sc.prompt({ requestId: 'rotate-seed-' + Date.now(), sessionId: createdId, mode: 'queue', content: [{ type: 'text', text: WAKE_TEXT }] }, FAKE_SIGNAL);
          S.seeded = true; S.seededFor = createdId; steps.push('wake:ok');
        } catch (e) { steps.push('wake:err'); noteError('rotate-wake', e) }
        steps.push('persist:' + (await saveState()));
        return { ok: true, archived: oldId, sessionId: createdId, sessionTitle: SESSION_TITLE, steps };
      };

        _TMP + '\\chat-feed\\panel.json',
        _TMP + '\\dsh-chat-digest\\panel.json',
      ];
      const PANEL_DEFAULT = { version: 1, fallbackKind: 'life', kindAliases: {}, sections: [
        { id: 'todo', title: '一、待办', kinds: ['todo'], order: 1 },
        { id: 'chance', title: '二、机会与招募', kinds: ['chance'], order: 2 },
        { id: 'info', title: '三、有用信息', kinds: ['official', 'resource', 'peer', 'life'], order: 3 },
      ] };
      const panelSections = (j) => {
        if (!j || !Array.isArray(j.sections) || !j.sections.length) return null;
        const seen = new Set();
        for (const s of j.sections) {
          if (!s || typeof s.id !== 'string' || typeof s.title !== 'string' || !Array.isArray(s.kinds)) return null;
          if (seen.has(s.id)) return null; seen.add(s.id);
        }
        return j.sections.slice().sort((a, b) => (a.order || 0) - (b.order || 0));
      };
      const readPanel = async () => {
        try {
          // ⚠ 不要用 fsSvc.stat（fs 服务只有 readText/writeText/resolve，stat 不存在 —— 实测踩过：
          //   一调用就抛 → 明明文件在也永远落 fallback）。小 JSON 直接读文本、比内容决定是否刷新缓存。
          const text = await fsSvc.readText(await fsSvc.resolve(PANEL_FILE));
          if (S.panelCacheText !== text) { S.panelCacheText = text; S.panelCache = JSON.parse(text); }
          const secs = panelSections(S.panelCache);
          S.panelErr = '';
          return secs ? { ok: true, source: 'panel.json(' + text.length + 'B)', fallback: false, json: S.panelCache, sections: secs }
                      : { ok: true, source: 'panel.json(结构非法→内置默认)', fallback: true, json: PANEL_DEFAULT, sections: PANEL_DEFAULT.sections };
        } catch (e) {
          // A25 一次性迁移 + 改名兼容：旧位置是**列表**，按顺序试
          //   （改名前的 %LOCALAPPDATA%\\chat-feed\\panel.json 才是他自定义的那份）。
          try {
            let oldList = [];
            if (typeof PANEL_OLD === 'string' && PANEL_OLD) oldList = [PANEL_OLD];
            else if (typeof PANEL_OLD !== 'undefined' && Array.isArray(PANEL_OLD)) oldList = PANEL_OLD;
            let oldText = '';
            for (const cand of oldList) {
              if (!cand) continue;
              try { const t = await fsSvc.readText(await fsSvc.resolve(cand)); if (t) { oldText = t; break } } catch (e3) {}
            }
            if (oldText) {
              if (S.panelCacheText !== oldText) { S.panelCacheText = oldText; S.panelCache = JSON.parse(oldText); }
              try { await fsSvc.writeText(await fsSvc.resolve(PANEL_FILE), oldText) } catch (e2) {}
              const secs0 = panelSections(S.panelCache);
              S.panelErr = '';
              return secs0 ? { ok: true, source: 'panel.json(' + oldText.length + 'B，已从旧 %TEMP% 迁到 ' + PANEL_FILE + ')', fallback: false, json: S.panelCache, sections: secs0 }
                           : { ok: true, source: 'panel.json(结构非法→内置默认)', fallback: true, json: PANEL_DEFAULT, sections: PANEL_DEFAULT.sections };
            }
          } catch (e2) { /* 旧位置也没有：本来就是全新安装，走内置默认 */ }
          S.panelErr = String((e && e.message) || e);
          return { ok: true, source: '内置默认', fallback: true, json: PANEL_DEFAULT, sections: PANEL_DEFAULT.sections };
        }
      };
      // 面板自证（面板扩展 spec §2.3）：`/api/panel` 与 `/state.panel` 共用同一份计算。
      // 主 agent 改完 `panel.json`（或 POST `/panel/set`）后，读任一个都能确认"生效没 / 有没有未知 kind"。
      const panelView = async () => {
        const r = await readPanel();
        const known = new Set();
        for (const s of r.sections) for (const k of s.kinds) known.add(k);
        for (const k of Object.keys(r.json.kindAliases || {})) known.add(k);
        const unknown = [];
        for (const it of (S.items || [])) if (it && it.kind && !known.has(it.kind)) unknown.push(String(it.kind));
        return { ok: true, source: r.source, fallback: r.fallback, err: S.panelErr || '', version: r.json.version || 1,
                 fallbackKind: r.json.fallbackKind || 'life', kindAliases: r.json.kindAliases || {},
                 sections: r.sections.map((s) => ({ id: s.id, title: s.title, kinds: s.kinds, order: s.order || 0,
                   count: (S.items || []).filter((x) => x && x.kind && s.kinds.indexOf(x.kind) >= 0).length })),
                 unknownKinds: Array.from(new Set(unknown)), itemCount: (S.items || []).length, fallbackUsed: 0 };
      };
      routeOk.panel = route('GET', 'panel', panelView);
      routeOk.panelSet = route('POST', 'panel/set', async (body) => {
        const j = body && (body.json || body);
        const secs = panelSections(j);
        if (!secs) return { ok: false, error: '校验失败：sections 必须是非空数组，且每项含 id(string)/title(string)/kinds(数组)，id 不得重复' };
        try {
          await fsSvc.writeText(await fsSvc.resolve(PANEL_FILE), JSON.stringify(j, null, 2));
          S.panelCache = j; S.panelMtime = null;      // 下次读盘刷新
          return { ok: true, sections: secs.length, source: PANEL_FILE };
        } catch (e) { noteError('panel-write', e); return { ok: false, error: String((e && e.message) || e) } }
      });
      routeOk.setAuto = route('POST', 'set-auto', async (body) => {
        if (body && typeof body.on === 'boolean') S.auto = body.on;
        if (body && typeof body.time === 'string' && /^[0-2]\d:[0-5]\d$/.test(body.time)) S.autoTime = body.time;
        const persist = await saveState();
        return { auto: S.auto, autoTime: S.autoTime, persist };
      });
      const failed = Object.keys(routeOk).filter((k) => routeOk[k] === false).concat(registered.ui ? [] : ['ui.js'], registered.right ? [] : ['right.js']);
      if (failed.length) { S.lastRouteError = '注册失败：' + failed.join(', ') + (S.lastRouteError ? ' | ' + S.lastRouteError : ''); console.error('[cf] 路由注册失败：', failed.join(', ')) }

      try {
        const off = ws.tapIndex((html) => {
          try {
            if (typeof html !== 'string') return html;
            let out = html;
            // 顺序：ui.js 先、right.js 后 —— right.js 只负责定义 window.__cfwRight，
            // ui.js 的 renderPanel() 才是真正 mount 它的地方；反过来会让后加载的脚本把前一份覆盖掉。
            if (out.indexOf(SCRIPT_UI) < 0) { const tag = '<script src="' + SCRIPT_UI + '" defer></script>'; out = out.indexOf('</body>') >= 0 ? out.replace('</body>', tag + '</body>') : out + tag }
            if (out.indexOf(SCRIPT_RIGHT) < 0) { const tag = '<script src="' + SCRIPT_RIGHT + '" defer></script>'; out = out.indexOf('</body>') >= 0 ? out.replace('</body>', tag + '</body>') : out + tag }
            return out;
          } catch (e) { return html }
        });
        disposers.push(off);
      } catch (e) { noteError('tapIndex', e) }

      ctx.effect(() => () => { disposers.forEach((off) => { try { off() } catch (e) {} }) });
      console.log('[cf] ready v34：主 agent 入口链指向规格；' + JSON.stringify(routeOk));
    } catch (e) { console.error('[cf] apply 失败：', e) }
  },
};
